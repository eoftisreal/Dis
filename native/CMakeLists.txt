cmake_minimum_required(VERSION 3.24)
project(DisCodec LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Dis.Codec requires Windows Media Foundation")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(DisCodec SHARED encoder.cpp)
set_target_properties(DisCodec PROPERTIES OUTPUT_NAME "Dis.Codec")
target_compile_definitions(DisCodec PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(DisCodec PRIVATE mfplat mfuuid mfreadwrite ole32 oleaut32 wmcodecdspuuid avrt)
add_executable(DisCodecSmoke smoke.cpp)
target_link_libraries(DisCodecSmoke PRIVATE DisCodec)
