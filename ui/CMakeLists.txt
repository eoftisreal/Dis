cmake_minimum_required(VERSION 3.24)
project(DisDesktop VERSION 0.4.4 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(Dis WIN32 main.cpp DesktopBridge.cpp DesktopBridge.h)
configure_file(resources/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
target_sources(Dis PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
file(GLOB qml_sources RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "qml/*.qml")
foreach(qml_source IN LISTS qml_sources)
    get_filename_component(qml_name "${qml_source}" NAME)
    set_source_files_properties("${qml_source}" PROPERTIES QT_RESOURCE_ALIAS "${qml_name}")
endforeach()
qt_add_qml_module(Dis URI Dis.Desktop VERSION 1.0
    QML_FILES qml/Main.qml qml/Theme.qml qml/Icon.qml qml/AppButton.qml qml/AppSwitch.qml
        qml/AppCombo.qml qml/SectionCard.qml qml/DisplayPage.qml
        qml/DevicesPage.qml qml/SettingsPage.qml qml/AboutPage.qml)
target_link_libraries(Dis PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network)
target_compile_definitions(Dis PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(Dis PRIVATE /utf-8)
endif()
